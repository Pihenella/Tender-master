> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Improve XLSX form filling accuracy by replacing blind row/col coordinates with a two-phase approach: Claude maps fields semantically, code applies values by cell addresses.

**Architecture:** New V2 pipeline runs alongside V1. `buildFormMap()` analyzes Excel structure into typed regions. Claude maps data fields to cell addresses. `resolveMapping()` programmatically substitutes values. `selfCheck()` verifies and retries. UI toggle switches between V1/V2.

**Tech Stack:** ExcelJS (existing), Vitest (existing), SvelteKit/Next.js hybrid, Convex backend, Claude CLI

**Critical note:** The actual running processor is `scripts/local-processor.mts` (1350+ lines, standalone polling script). `src/lib/server/processor.ts` is a secondary/server version. Both must be updated, but `local-processor.mts` is the primary target.

// src/lib/server/formMap.ts
import ExcelJS from "exceljs";

export interface CellRef {
  cell: string;
  value: string;
}

export interface TableColumn {
  col: string;
  header: string;
}

export type Region =
  | { type: "header"; range: string; value: string }
  | { type: "field"; label: CellRef; input: CellRef }
  | {
      type: "table";
      headerRow: number;
      columns: TableColumn[];
      dataStartRow: number;
      existingRows: number;
    }
  | { type: "static"; cell: string; value: string };

export interface FormMap {
  sheet: string;
  dimensions: string;
  mergedCells: string[];
  regions: Region[];
}

function cellValue(cell: ExcelJS.Cell): string {
  const v = cell.value;
  if (v === null || v === undefined) return "";
  if (typeof v === "object" && "result" in v) return String((v as any).result);
  if (typeof v === "object" && "text" in v) return String((v as any).text);
  if (typeof v === "object" && "richText" in v) {
    return (v as any).richText.map((r: any) => r.text).join("");
  }
  return String(v);
}

function colLetter(colNumber: number): string {
  let result = "";
  let n = colNumber;
  while (n > 0) {
    n--;
    result = String.fromCharCode(65 + (n % 26)) + result;
    n = Math.floor(n / 26);
  }
  return result;
}

export async function buildFormMap(buffer: Buffer): Promise<FormMap> {
  const wb = new ExcelJS.Workbook();
  await wb.xlsx.load(buffer as unknown as ExcelJS.Buffer);
  const ws = wb.worksheets[0];
  if (!ws) throw new Error("No worksheets found");

  const merges: string[] = (ws.model.merges as string[]) || [];

  const formMap: FormMap = {
    sheet: ws.name,
    dimensions: ws.dimensions?.toString() || "",
    mergedCells: merges,
    regions: [],
  };

  // Track which cells are already classified
  const classified = new Set<string>();

  // Pass 1: Detect merged cell headers
  for (const merge of merges) {
    const [start] = merge.split(":");
    const val = cellValue(ws.getCell(start));
    if (val.trim()) {
      formMap.regions.push({ type: "header", range: merge, value: val.trim() });
      // Mark all cells in range as classified
      const [startCell, endCell] = merge.split(":");
      const startMatch = startCell.match(/([A-Z]+)(\d+)/);
      const endMatch = endCell.match(/([A-Z]+)(\d+)/);
      if (startMatch && endMatch) {
        for (let r = parseInt(startMatch[2]); r <= parseInt(endMatch[2]); r++) {
          for (
            let c = startMatch[1].charCodeAt(0) - 64;
            c <= endMatch[1].charCodeAt(0) - 64;
            c++
          ) {
            classified.add(`${colLetter(c)}${r}`);
          }
        }
      }
    }
  }

  // Pass 2: Detect tables (3+ consecutive non-empty cells in a row)
  const rowCount = ws.rowCount;
  for (let r = 1; r <= rowCount; r++) {
    const row = ws.getRow(r);
    const nonEmptyCells: { col: number; value: string }[] = [];

    row.eachCell({ includeEmpty: false }, (cell, colNum) => {
      const addr = `${colLetter(colNum)}${r}`;
      if (classified.has(addr)) return;
      const val = cellValue(cell);
      if (val.trim()) {
        nonEmptyCells.push({ col: colNum, value: val.trim() });
      }
    });

    if (nonEmptyCells.length >= 3) {
      // Check if next row is mostly empty (table data area)
      const nextRow = ws.getRow(r + 1);
      let nextRowEmpty = true;
      nextRow.eachCell({ includeEmpty: false }, (cell) => {
        if (cellValue(cell).trim()) nextRowEmpty = false;
      });

      const columns: TableColumn[] = nonEmptyCells.map((c) => ({
        col: colLetter(c.col),
        header: c.value,
      }));

      // Count existing data rows
      let existingRows = 0;
      if (!nextRowEmpty) {
        for (let dr = r + 1; dr <= rowCount; dr++) {
          const dataRow = ws.getRow(dr);
          let hasData = false;
          dataRow.eachCell({ includeEmpty: false }, (cell) => {
            if (cellValue(cell).trim()) hasData = true;
          });
          if (hasData) existingRows++;
          else break;
        }
      }

      formMap.regions.push({
        type: "table",
        headerRow: r,
        columns,
        dataStartRow: r + 1,
        existingRows,
      });

      // Mark header and data rows as classified
      for (const c of nonEmptyCells) {
        classified.add(`${colLetter(c.col)}${r}`);
      }
      for (let dr = r + 1; dr <= r + existingRows; dr++) {
        ws.getRow(dr).eachCell({ includeEmpty: false }, (_, colNum) => {
          classified.add(`${colLetter(colNum)}${dr}`);
        });
      }
    }
  }

  // Pass 3: Detect field pairs (label + adjacent empty/placeholder input)
  for (let r = 1; r <= rowCount; r++) {
    const row = ws.getRow(r);
    row.eachCell({ includeEmpty: false }, (cell, colNum) => {
      const addr = `${colLetter(colNum)}${r}`;
      if (classified.has(addr)) return;

      const val = cellValue(cell);
      if (!val.trim()) return;

      // Check right neighbor
      const rightAddr = `${colLetter(colNum + 1)}${r}`;
      if (classified.has(rightAddr)) return;

      const rightCell = ws.getCell(rightAddr);
      const rightVal = cellValue(rightCell);

      // Input cell: empty, or contains underscores/placeholder
      const isInput =
        rightVal === "" ||
        /^[_\s.…-]+$/.test(rightVal) ||
        /^(указать|заполнить|ввести)/i.test(rightVal);

      if (isInput) {
        formMap.regions.push({
          type: "field",
          label: { cell: addr, value: val.trim() },
          input: { cell: rightAddr, value: rightVal },
        });
        classified.add(addr);
        classified.add(rightAddr);
      }
    });
  }

  // Pass 4: Remaining non-empty cells → static
  for (let r = 1; r <= rowCount; r++) {
    ws.getRow(r).eachCell({ includeEmpty: false }, (cell, colNum) => {
      const addr = `${colLetter(colNum)}${r}`;
      if (classified.has(addr)) return;
      const val = cellValue(cell);
      if (val.trim()) {
        formMap.regions.push({ type: "static", cell: addr, value: val.trim() });
        classified.add(addr);
      }
    });
  }

  return formMap;
}

// src/lib/server/fillV2.ts
import ExcelJS from "exceljs";
import type { FormMap } from "./formMap";

export interface MappingEntry {
  cell: string;
  dataPath: string;
  confidence: "high" | "medium" | "low";
}

export interface ComputedField {
  cell: string;
  expression: string;
  label: string;
}

export interface ClaudeMapping {
  mappings: MappingEntry[];
  tables: Array<{
    dataStartRow: number;
    columnMap: Record<string, string>;
  }>;
  unmapped: string[];
  computed: ComputedField[];
}

export interface CellValue {
  cell: string;
  value: string | number;
}

export interface ResolveResult {
  cellValues: CellValue[];
  tableData: Array<{
    startRow: number;
    rows: Record<string, string | number>[];
  }>;
  unresolved: string[];
}

export interface CheckResult {
  applied: number;
  failed: number;
  missing: number;
  details: Array<{
    cell: string;
    expected: string;
    actual: string;
    reason: "merged_cell" | "write_failed" | "protected" | "unmapped";
  }>;
}

function getByPath(obj: any, path: string): any {
  return path.split(".").reduce((o, key) => {
    if (o === null || o === undefined) return undefined;
    return o[key];
  }, obj);
}

function resolveArrayPath(
  obj: any,
  path: string
): { basePath: string; field: string; values: any[] } | null {
  const match = path.match(/^(.+?)\[\]\.(.+)$/);
  if (!match) return null;
  const [, basePath, field] = match;
  const arr = getByPath(obj, basePath);
  if (!Array.isArray(arr)) return null;
  return { basePath, field, values: arr.map((item) => getByPath(item, field)) };
}

export function resolveMapping(
  mapping: ClaudeMapping,
  context: any
): ResolveResult {
  const cellValues: CellValue[] = [];
  const unresolved: string[] = [];
  const tableData: ResolveResult["tableData"] = [];

  // Resolve field mappings
  for (const m of mapping.mappings) {
    if (m.confidence === "low") {
      unresolved.push(m.cell);
      continue;
    }
    const value = getByPath(context, m.dataPath);
    if (value !== undefined && value !== null) {
      cellValues.push({ cell: m.cell, value: value });
    } else {
      unresolved.push(m.cell);
    }
  }

  // Resolve tables
  for (const table of mapping.tables) {
    // Find the first array path to determine row count
    const arrayPaths = Object.values(table.columnMap).filter((p) =>
      p.includes("[]")
    );
    if (arrayPaths.length === 0) continue;

    const firstArray = resolveArrayPath(context, arrayPaths[0]);
    if (!firstArray) continue;

    const rowCount = firstArray.values.length;
    const rows: Record<string, string | number>[] = [];

    for (let i = 0; i < rowCount; i++) {
      const row: Record<string, string | number> = {};
      for (const [col, dataPath] of Object.entries(table.columnMap)) {
        if (dataPath === "rowNumber") {
          row[col] = i + 1;
        } else if (dataPath.includes("[]")) {
          const resolved = resolveArrayPath(context, dataPath);
          if (resolved && i < resolved.values.length) {
            row[col] = resolved.values[i] ?? "";
          }
        } else {
          const val = getByPath(context, dataPath);
          row[col] = val ?? "";
        }
      }
      rows.push(row);
    }

    tableData.push({ startRow: table.dataStartRow, rows });
  }

  // Add unmapped cells
  unresolved.push(...mapping.unmapped);

  return { cellValues, tableData, unresolved };
}

function colToNumber(col: string): number {
  let result = 0;
  for (let i = 0; i < col.length; i++) {
    result = result * 26 + (col.charCodeAt(i) - 64);
  }
  return result;
}

function getMasterCell(ws: ExcelJS.Worksheet, cellAddr: string): string {
  const merges = (ws.model.merges as string[]) || [];
  for (const merge of merges) {
    const [start, end] = merge.split(":");
    const startMatch = start.match(/([A-Z]+)(\d+)/);
    const endMatch = end.match(/([A-Z]+)(\d+)/);
    const cellMatch = cellAddr.match(/([A-Z]+)(\d+)/);
    if (!startMatch || !endMatch || !cellMatch) continue;

    const sCol = colToNumber(startMatch[1]);
    const eCol = colToNumber(endMatch[1]);
    const sRow = parseInt(startMatch[2]);
    const eRow = parseInt(endMatch[2]);
    const cCol = colToNumber(cellMatch[1]);
    const cRow = parseInt(cellMatch[2]);

    if (cCol >= sCol && cCol <= eCol && cRow >= sRow && cRow <= eRow) {
      return start;
    }
  }
  return cellAddr;
}

export async function applyXlsxV2(
  buffer: Buffer,
  cellValues: CellValue[],
  tableData?: Array<{ startRow: number; rows: Record<string, string | number>[] }>
): Promise<Buffer> {
  const wb = new ExcelJS.Workbook();
  await wb.xlsx.load(buffer as unknown as ExcelJS.Buffer);
  const ws = wb.worksheets[0];
  if (!ws) return buffer;

  // Write cell values
  for (const { cell, value } of cellValues) {
    const masterAddr = getMasterCell(ws, cell);
    ws.getCell(masterAddr).value = value as ExcelJS.CellValue;
  }

  // Write table rows
  if (tableData) {
    for (const { startRow, rows } of tableData) {
      for (let i = 0; i < rows.length; i++) {
        const wsRow = ws.getRow(startRow + i);
        for (const [col, value] of Object.entries(rows[i])) {
          const colNum = colToNumber(col);
          wsRow.getCell(colNum).value = value as ExcelJS.CellValue;
        }
        wsRow.commit();
      }
    }
  }

  return Buffer.from(await wb.xlsx.writeBuffer());
}

export async function selfCheck(
  filledBuffer: Buffer,
  expectedValues: CellValue[],
  formMap: FormMap
): Promise<CheckResult> {
  const wb = new ExcelJS.Workbook();
  await wb.xlsx.load(filledBuffer as unknown as ExcelJS.Buffer);
  const ws = wb.worksheets[0];

  const result: CheckResult = {
    applied: 0,
    failed: 0,
    missing: 0,
    details: [],
  };

  if (!ws) {
    result.failed = expectedValues.length;
    return result;
  }

  for (const { cell, value } of expectedValues) {
    const masterAddr = getMasterCell(ws, cell);
    const actual = ws.getCell(masterAddr).value;
    const expected = String(value);
    const actualStr = actual === null || actual === undefined ? "" : String(actual);

    if (actualStr === expected || actual === value) {
      result.applied++;
    } else {
      result.failed++;
      const isMerged = masterAddr !== cell;
      result.details.push({
        cell,
        expected,
        actual: actualStr,
        reason: isMerged ? "merged_cell" : "write_failed",
      });
    }
  }

  // Check for unfilled input fields from formMap
  for (const region of formMap.regions) {
    if (region.type !== "field") continue;
    const inputAddr = region.input.cell;
    const wasTargeted = expectedValues.some((v) => v.cell === inputAddr);
    if (wasTargeted) continue;

    const masterAddr = getMasterCell(ws, inputAddr);
    const val = ws.getCell(masterAddr).value;
    const valStr = val === null || val === undefined ? "" : String(val);
    const isEmpty =
      valStr === "" || /^[_\s.…-]+$/.test(valStr);

    if (isEmpty) {
      result.missing++;
      result.details.push({
        cell: inputAddr,
        expected: "(unfilled)",
        actual: valStr,
        reason: "unmapped",
      });
    }
  }

  return result;
}
